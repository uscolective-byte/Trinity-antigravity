function formatMarkdown(text){
  if(!text)return '';
  const d=document.createElement('div');
  d.textContent=text;
  let s=d.innerHTML;
  s=s.replace(/```([a-zA-Z0-9_-]*)\n([\s\S]*?)```/g,'<pre class="code-block"><code class="lang-$1">$2</code></pre>');
  s=s.replace(/`([^`\n]+)`/g,'<code class="inline-code">$1</code>');
  s=s.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
  s=s.replace(/(^|[^*])\*([^*\n]+)\*/g,'$1<em>$2</em>');
  s=s.replace(/^[ \t]*[\*\-][ \t]+(.+)$/gm,'<li>$1</li>');
  s=s.replace(/(<li>[\s\S]*?<\/li>)/g,'<ul class="md-list">$1</ul>');
  s=s.replace(/<\/ul>\s*<ul class="md-list">/g,'');
  s=s.replace(/\n\n+/g,'<br><br>');
  return s;
}
function renderReply(target,text){

  const match=text.match(/\/api\/assistant\/media\/images\/[0-9a-f-]{36}/i);
  if(match){
    const image=document.createElement('img');
    image.className='generated-media';
    image.src=match[0];
    image.alt='Obrázok vytvorený Trinity';
    image.loading='lazy';
    target.append(image);
  }
  const fileMatches=Array.from(text.matchAll(/\/api\/assistant\/files\/([0-9a-f-]{36})\/([a-zA-Z0-9._\-]+)/gi));
  if(fileMatches.length){
    for(const fm of fileMatches){
      const url=fm[0];
      const fname=fm[2];
      const ext=fname.split('.').pop()?.toLowerCase();
      let icon='📄';
      if(ext==='pdf')icon='📕';
      else if(ext==='py')icon='🐍';
      else if(ext==='json')icon='📦';
      else if(['js','ts','mjs'].includes(ext))icon='⚡';
      else if(['jpg','png','webp','gif','svg'].includes(ext))icon='🖼️';
      else if(ext==='csv')icon='📊';
      const fileCard=document.createElement('div');
      fileCard.className='generated-file-card';
      fileCard.innerHTML=`<span class="file-icon">${icon}</span><div class="file-info"><b>${fname}</b><small>Vytvorený súbor v R2 archíve</small></div><a href="${url}?download=1" download="${fname}" class="file-download-btn">⬇ Stiahnuť</a><a href="${url}" target="_blank" rel="noopener" class="file-view-btn">Otvoriť ↗</a>`;
      target.append(fileCard);
    }
  }
}

function renderSelfImprovement(data){
  const status=$('self-improvement-deploy-status'),list=$('self-improvement-list');
  status.textContent=data.capabilities?.deployment||'Nasadenie sa riadi GitHub Actions.';
  list.replaceChildren();
  if(!data.items?.length){list.append(node('p','Zatiaľ nie sú evidované žiadne návrhy.','skeleton'));return;}
  for(const item of data.items){
    const card=node('article',undefined,'api-key-item'),info=node('div');
    info.append(node('b',item.title),node('small',`${item.kind} · ${item.status}`),node('p',item.description));
    card.append(info);
    if(item.plugin_id)card.append(node('small',`Modul: ${item.plugin_id}`));
    if(item.status==='pending'&&data.can_approve===true){
      const approve=node('button','Schváliť návrh','primary');approve.type='button';
      approve.onclick=safe(async()=>{await api('/self-improvement/'+item.id+'/approve',{});await loadSettings();notice('Návrh bol schválený a zapísaný do auditu.');});
      card.append(approve);
    }
    if(item.status==='approved'&&!item.result_json){
      const apply=node('button',item.kind==='plugin_install'?'Nainštalovať modul':'Vykonať návrh','primary');apply.type='button';
      apply.onclick=safe(async()=>{await api('/self-improvement/'+item.id+'/apply',{});await loadSettings();notice('Návrh bol spracovaný. Výsledok a prípadné zostávajúce kroky sú pri návrhu.');});
      card.append(apply);
    }
    if(item.result_json){info.append(node('small',item.result_json.status||'Spracované'));if(item.result_json.url){const link=node('a','Otvoriť výsledok');link.href=item.result_json.url;link.target='_blank';link.rel='noopener noreferrer';info.append(link);}if(item.result_json.error)info.append(node('p',item.result_json.error));}
    list.append(card);
  }
}

function slugifyId(text){return String(text||'ponuka').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/[^a-z0-9]+/g,'-').replace(/^-+|-+$/g,'').slice(0,70)||'ponuka';}
function filterAndRenderDomains(){
  let items=cachedDomains||[];
  if(domainFilter!=='all')items=items.filter(d=>d.status===domainFilter);
  if(domainSearch)items=items.filter(d=>(d.domain_name||'').toLowerCase().includes(domainSearch)||(d.registrar||'').toLowerCase().includes(domainSearch)||(d.notes||'').toLowerCase().includes(domainSearch));
  renderDomains(items);
  renderDomainMarketplace();
}
function renderDomainStats(){
  const all=cachedDomains||[],set=(id,v)=>{const el=$(id);if(el)el.textContent=String(v);};
  set('dstat-total',all.length);
  set('dstat-active',all.filter(d=>d.status==='active').length);
  set('dstat-sale',all.filter(d=>d.status==='for_sale'||d.listing_type==='sale'||d.listing_type==='both').length);
  set('dstat-lease',all.filter(d=>d.status==='leased'||d.listing_type==='lease'||d.listing_type==='both').length);
  const domainVal=all.reduce((sum,d)=>sum+(Number(d.sale_price)||0)+((Number(d.lease_price_monthly)||0)*12),0);
  set('dstat-value',new Intl.NumberFormat('sk-SK',{style:'currency',currency:'EUR',maximumFractionDigits:0}).format(domainVal));
}

let lastUiUpdatedAt=null,cachedUiEvo=null,docCategoryFilter='',docSearchQuery='';
function applyUiDomCustomizations(st){
  if(!st)return;
  const brandLink=document.querySelector('.sidebar .brand > span:last-child');
  if(brandLink&&st.brand_title){
    const sub=st.brand_subtitle||'Tvoja osobná asistentka';
    brandLink.innerHTML='';
    brandLink.append(document.createTextNode(st.brand_title),node('small',sub));
  }
  const wHeading=document.querySelector('#welcome h1');
  if(wHeading&&st.welcome_heading)wHeading.textContent=st.welcome_heading;
  const wSub=document.querySelector('#welcome p');
  if(wSub&&st.welcome_subtext)wSub.textContent=st.welcome_subtext;
  const slot=$('ui-custom-widgets-slot');
  if(slot){
    const widgets=Array.isArray(st.widgets)?st.widgets:(Array.isArray(st.custom_widgets)?st.custom_widgets:[]);
    slot.replaceChildren();
    slot.hidden=!widgets.length;
    for(const w of widgets){
      const card=node('section',undefined,'panel ui-custom-widget-card');
      const head=node('div',undefined,'panel-head');
      const left=node('div');
      left.append(node('span',w.badge||'TRINITY UI WIDGET','eyebrow'),node('h2',w.title||'Vlastný panel'));
      head.append(left,node('span','LIVE UI','detail-chip'));
      card.append(head,node('p',w.content||w.body||''));
      slot.append(card);
    }
  }
}
async function refreshUiEvolutionLive(forceReloadCss=false){
  try{
    const data=await api('/ui-evolution');
    cachedUiEvo=data;
    const st=data.state||{};
    applyUiDomCustomizations(st);
    if(forceReloadCss||(st.updated_at&&st.updated_at!==lastUiUpdatedAt)){
      lastUiUpdatedAt=st.updated_at;
      const themeLink=document.querySelector('link[href^="/theme.css"]');
      if(themeLink)themeLink.href='/theme.css?v='+Date.now();
    }
    return data;
  }catch{return null;}
}

async function loadDomainEnvironments(){

      api('/domains/environments'),

    ]);
    const studioSelect=$('domain-env-studio-project');
    if(studioSelect){

    box.replaceChildren(node('p','Načítanie prostredí vyžaduje prihlásenie.','skeleton'));
  }
}
function renderDomainEnvironments(data){
  const box=$('domain-environments-list');if(!box)return;
  const items=data?.items||[];
  if($('domain-env-count'))$('domain-env-count').textContent=`${items.length} domén s prostrediami`;
  box.replaceChildren();
  if(!items.length){
    const empty=node('div',undefined,'domain-market-empty');
    empty.append(
      node('b','Zatiaľ nemáš vytvorené žiadne doménové prostredie (Production / Staging / Dev).'),
      node('small','Vyplň vyššie názov domény (napr. moj-projekt.sk), vyber šablónu a klikni na „🚀 Vytvoriť doménu, prostredia & spustiť produkciu“.')
    );
    box.append(empty);
    return;
  }
  for(const item of items){
    const card=node('article',undefined,'domain-market-card domain-env-card');
    const head=node('div',undefined,'domain-market-head');
    head.append(node('strong',item.domain_name),node('span',`SSL: ${item.ssl_mode||'strict'} · Proxy ON`,'domain-badge active'));
    const envGrid=node('div',undefined,'domain-env-badges');
    const envs=item.environments||{};
    for(const envKey of ['production','staging','development']){
      const eInfo=envs[envKey];
      if(!eInfo)continue;
      const pill=node('div',undefined,'domain-env-row');
      const isLive=eInfo.status==='deployed'||eInfo.status==='live';
      const statusText=isLive?`LIVE · v${eInfo.version||1}`:'Pripravené';
      pill.append(
        node('b',`${envKey.toUpperCase()}: `),
        node('span',statusText,'price-tag '+(isLive?'sale':'lease'))
      );
      const livePath=eInfo.live_path||eInfo.public_path||`/sites/${item.domain_name}/${envKey}`;
      if(livePath){
        const link=node('a',livePath,'button-link');
        link.href=livePath;link.target='_blank';link.rel='noopener noreferrer';
        pill.append(link);
      }
      const deployBtn=node('button',isLive?'↻ Preprodukovať':'⚡ Nasadiť');
      deployBtn.type='button';
      deployBtn.onclick=safe(async()=>{
        const studioId=$('domain-env-studio-project')?.value;
        const tpl=studioId?'studio_project':($('domain-env-template')?.value||'business_landing');
        const res=await api('/domains/environments/deploy',{
          domain_name:item.domain_name,
          environment:envKey,
          template:tpl,
          title:$('domain-env-title')?.value.trim()||item.domain_name,
          subtitle:$('domain-env-subtitle')?.value.trim()||undefined,
          studio_project_id:studioId||undefined
        });
        notice(`Prostredie ${envKey} pre ${item.domain_name} bolo vyprodukované na ${res.environment?.live_path||livePath} (v${res.environment?.version||1}).`);
        await loadDomainEnvironments();
      });
      pill.append(deployBtn);
      envGrid.append(pill);
    }
    card.append(head,envGrid);
    box.append(card);
  }
}

async function loadTrinityDocuments(){
  const list=$('trinity-doc-list');if(!list)return;
  try{
    const q=new URLSearchParams();
    if(docCategoryFilter)q.set('category',docCategoryFilter);
    if(docSearchQuery)q.set('search',docSearchQuery);
    q.set('limit','60');
    const data=await api('/documents?'+q.toString());
    renderTrinityDocuments(data);
  }catch(e){
    list.replaceChildren(node('p','Načítavanie Trinity Dokumentov zlyhalo: '+e.message,'skeleton'));
  }
}
function renderTrinityDocuments(data){
  const items=data?.items||[];
  const set=(id,v)=>{const el=$(id);if(el)el.textContent=String(v);};
  set('docstat-total',data?.total??items.length);
  set('docstat-steps',items.filter(d=>d.category==='workflow_step').length);
  set('docstat-ui',items.filter(d=>d.category==='ui_evolution'||d.category==='self_improvement').length);
  set('docstat-domains',items.filter(d=>d.category==='domain_production').length);
  const list=$('trinity-doc-list');if(!list)return;
  list.replaceChildren();
  if(!items.length){
    list.append(node('p','Zatiaľ nie sú žiadne Trinity Dokumenty pre zvolený filter. Každý krok v chate, úprava UI alebo vytvorenie domény automaticky vytvorí nový dokument.','skeleton'));
    return;
  }
  for(const doc of items){
    const card=node('article',undefined,'api-key-item trinity-doc-card');
    const info=node('div',undefined,'commerce-offer-info');
    const top=node('div',undefined,'commerce-offer-top');
    top.append(node('b',doc.title),node('span',doc.category,'commerce-provider-badge revolut-pro'));
    if(doc.step_name)top.append(node('span',doc.step_name,'detail-chip'));
    info.append(top);
    if(doc.summary)info.append(node('p',doc.summary));
    info.append(node('small',`ID: ${doc.id} · Autor: ${doc.actor||doc.author||'trinity'} · ${new Date(doc.created_at).toLocaleString('sk-SK')}`));
    const actions=node('div',undefined,'commerce-offer-actions');
    const readBtn=node('button','👁 Čítať detail','primary');
    readBtn.type='button';
    readBtn.onclick=safe(()=>openTrinityDocReader(doc.id));
    const mdLink=node('a','⬇ .MD','button-link');
    mdLink.href=`/api/assistant/documents/${doc.id}/export?format=md`;
    const pdfLink=node('a','📕 .PDF','button-link');
    pdfLink.href=`/api/assistant/documents/${doc.id}/export?format=pdf`;
    const delBtn=node('button','Zmazať','danger');
    delBtn.type='button';
    delBtn.onclick=safe(async()=>{

      notice('Trinity Dokument bol odstránený.');
    });

}
async function openTrinityDocReader(docId){
